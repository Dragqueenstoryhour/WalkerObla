  const playTTS = async (item: ProcessedItem) => {
    try {
      const state = getRecordingState(item.id);
      const textToSpeak = item.text;

      // Construct the SSML string
      let ssmlText = `<speak version="1.0" xmlns="http://www.w3.org/2001/10/synthesis" xml:lang="en-US">`;
      ssmlText += `<voice name="en-US-AvaNeural">`; // Specify the Azure AI Speech native voice

      if (state.slowPlayback) {
        // Use SSML prosody rate to slow down to 60%
        // Using "60%" is clear and robust. Your observation of "0.6" speeding it up
        // was likely due to an engine interpreting it as a multiplier (e.g., 0.6x normal speed)
        // rather than a percentage, resulting in an extremely fast playback.
        ssmlText += `<prosody rate="60%">`;
        ssmlText += textToSpeak;
        ssmlText += `</prosody>`;
      } else {
        ssmlText += textToSpeak;
      }
      ssmlText += `</voice>`;
      ssmlText += `</speak>`;

      const response = await fetch("/api/speech/synthesize", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          ssml: ssmlText, // Send the SSML string instead of plain text and speed
          // You no longer need 'voice' or 'speed' parameters here directly,
          // as they are handled within the SSML.
        }),
      });

      if (!response.ok) throw new Error("Failed to generate speech");

      const audioBlob = await response.blob();
      const audioUrl = URL.createObjectURL(audioBlob);

      if (audioRef.current) {
        audioRef.current.pause();
        audioRef.current.currentTime = 0;
      }

      audioRef.current = new Audio(audioUrl);
      audioRef.current.play().catch(error => {
        console.error("Error playing TTS:", error);
      });

      audioRef.current.onended = () => {
        URL.revokeObjectURL(audioUrl);
      };
    } catch (error) {
      console.error("Error with text-to-speech:", error);
    }
  };
