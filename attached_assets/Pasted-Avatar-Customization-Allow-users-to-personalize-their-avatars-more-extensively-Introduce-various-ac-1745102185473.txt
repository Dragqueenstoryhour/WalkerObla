Example Code Snippet for Avatar Preview
Here's an example of how you might implement an avatar preview section:

import { useState } from 'react';
import { Avatar } from './Avatar';
const AvatarSelector = () => {
  const [selectedAvatar, setSelectedAvatar] = useState('default');
  const handleAvatarChange = (newAvatar) => {
    setSelectedAvatar(newAvatar);
  };
  return (
    <div className="avatar-selector">
      <h3>Select Your Avatar</h3>
      <div className="avatar-preview">
        <Avatar character={selectedAvatar} />
      </div>
      <div className="avatar-options">
        {['chicken', 'penguin', 'frog', 'tiger', 'monkey'].map((character) => (
          <button key={character} onClick={() => handleAvatarChange(character)}>
            {character}
          </button>
        ))}
      </div>
    </div>
  );
};
